\Table authors {
  id           int [pk, increment]
  full_name    varchar(100) [not null]
  email        varchar(100) [unique, not null]
}

Table documents {
  id           int [pk, increment]
  title        varchar(255) [not null]  
  type         varchar(10) [not null]     
  description  text      
  date_created date [not null]    
  last_edit    date            
  status_id    int [ref: > statuses.id]
}

Table statuses {
  id           int [pk, increment]
  name         varchar(50) [not null]           // например: "open", "closed"
}

Table applications {
  id           int [pk, increment]
  title        varchar(255) [not null]  
  status_id    int [ref: > statuses.id] 
  created_at   timestamp [not null]
}

Table document_authors {
  document_id  int [ref: > documents.id]
  author_id    int [ref: > authors.id]
  indexes {
    (document_id, author_id) [pk]
  }
}

Table application_documents {
  application_id int [ref: > applications.id]
  document_id    int [ref: > documents.id]
  indexes {
    (application_id, document_id) [pk]
  }
}
